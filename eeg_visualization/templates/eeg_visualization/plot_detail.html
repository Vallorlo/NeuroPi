import mne
import numpy as np
import matplotlib.pyplot as plt

# Load data
{% if plot_type == 'interactive' %}
raw = mne.io.read_raw_fif('sample_data.fif', preload=True)

{% elif plot_type == 'topographic' %}
evoked = mne.read_evokeds('sample_data-ave.fif')[0]
evoked.plot_topomap(times=[0.1, 0.2, 0.3], ch_type='eeg')
{% elif plot_type == 'timefreq' %}

freqs = np.arange(4, 40, 2)
tfr = mne.time_frequency.tfr_morlet(epochs, freqs, n_cycles=2)
tfr.plot(picks='Cz', baseline=(-0.5, 0))
{% elif plot_type == '3d' %}

fig = mne.viz.plot_alignment(raw.info, subject='sample',
                            surfaces='head-dense')
{% elif plot_type == 'ica' %}
ica = mne.preprocessing.ICA(n_components=20)
ica.fit(raw)
ica.plot_components()
{% elif plot_type == 'stats' %}

plot_connectivity_circle(con, node_names, title='Brain Connectivity')
