```python
def check_signal_quality(eeg_data):
    import numpy as np
    
    # حساب الانحراف المعياري لكل قناة
    std_values = np.std(eeg_data, axis=0)
    
    # التحقق من القنوات الضعيفة
    weak_channels = []
    for i, std_val in enumerate(std_values):
        if std_val < 5 or std_val > 100:  # قيم غير طبيعية
            weak_channels.append(f"Channel_{i}")
    
    if weak_channels:
        print(f"⚠️ قنوات ضعيفة: {weak_channels}")
    else:
        print("✅ جودة الإشارة جيدة")
```

```python
# المشكلة: نفاد الذاكرة أثناء معالجة البيانات الكبيرة
MemoryError: Unable to allocate array

# الحلول:
def process_large_data_efficiently(file_path):
    import pandas as pd
    import numpy as np
    
    # 1. قراءة البيانات على دفعات
    chunk_size = 10000
    processed_chunks = []
    
    for chunk in pd.read_csv(file_path, chunksize=chunk_size):
        # معالجة كل دفعة على حدة
        processed_chunk = preprocess_chunk(chunk)
        processed_chunks.append(processed_chunk)
    
    # دمج النتائج
    final_data = np.concatenate(processed_chunks, axis=0)
    return final_data

def preprocess_chunk(chunk):
    # معالجة مبسطة لتوفير الذاكرة
    return chunk.values.astype(np.float32)  # استخدام float32 بدلاً من float64
```

#### خطأ: "Invalid data format"
```python
# المشكلة: تنسيق البيانات غير صحيح
ValueError: Invalid data format for EEG processing

# الحل:
def validate_and_fix_data(data):
    import pandas as pd
    import numpy as np
    
    # التحقق من الأعمدة المطلوبة
    required_columns = ['AF3', 'F7', 'F3', 'FC5', 'T7', 'P7', 'O1', 
                       'O2', 'P8', 'T8', 'FC6', 'F4', 'F8', 'AF4']
    
    missing_columns = [col for col in required_columns if col not in data.columns]
    if missing_columns:
        print(f"❌ أعمدة مفقودة: {missing_columns}")
        return None
    
    # التحقق من القيم المفقودة
    if data.isnull().any().any():
        print("⚠️ يوجد قيم مفقودة، سيتم ملؤها")
        data = data.fillna(method='forward')
    
    # التحقق من القيم الشاذة
    for col in required_columns:
        q1 = data[col].quantile(0.25)
        q3 = data[col].quantile(0.75)
        iqr = q3 - q1
        lower_bound = q1 - 1.5 * iqr
        upper_bound = q3 + 1.5 * iqr
        
        outliers = data[(data[col] < lower_bound) | (data[col] > upper_bound)]
        if len(outliers) > 0:
            print(f"⚠️ قيم شاذة في {col}: {len(outliers)} قيمة")
    
    print("✅ البيانات صالحة للمعالجة")
    return data
```
